    <script>
        // State Management
        let balance = 0;
        let totalEarned = 0;
        let tasksCompleted = 0;
        let isAdWatching = false;

        // Initialize state from LocalStorage if exists
        window.onload = function() {
            const savedBalance = localStorage.getItem('ee_balance');
            const savedTotal = localStorage.getItem('ee_total');
            const savedTasks = localStorage.getItem('ee_tasks');

            if(savedBalance) balance = parseInt(savedBalance);
            if(savedTotal) totalEarned = parseInt(savedTotal);
            if(savedTasks) tasksCompleted = parseInt(savedTasks);

            updateUI();
        };

        function updateUI() {
            document.getElementById('userBalance').innerText = balance;
            document.getElementById('totalEarned').innerText = totalEarned;
            document.getElementById('tasksCount').innerText = tasksCompleted;

            localStorage.setItem('ee_balance', balance);
            localStorage.setItem('ee_total', totalEarned);
            localStorage.setItem('ee_tasks', tasksCompleted);
        }

        // Watch Ads System with Timer
        function startWatchAd() {
            if(isAdWatching) return;
            isAdWatching = true;

            // Open Sponsor Ad link (Replace with your actual Adsterra Direct Link)
            window.open('https://www.google.com', '_blank');

            document.getElementById('adReadyBox').classList.add('hidden');
            document.getElementById('adTimerBox').classList.remove('hidden');

            let timeLeft = 10;
            const timerCount = document.getElementById('timerCount');
            const timerProgress = document.getElementById('timerProgress');

            timerCount.innerText = timeLeft;
            timerProgress.style.width = '100%';

            const interval = setInterval(() => {
                timeLeft--;
                timerCount.innerText = timeLeft;
                timerProgress.style.width = (timeLeft * 10) + '%';

                if(timeLeft <= 0) {
                    clearInterval(interval);
                    // Add Reward
                    balance += 10;
                    totalEarned += 10;
                    tasksCompleted += 1;
                    updateUI();

                    // Reset UI
                    document.getElementById('adTimerBox').classList.add('hidden');
                    document.getElementById('adReadyBox').classList.remove('hidden');
                    isAdWatching = false;

                    alert("Success! Rs. 10 added to your balance.");
                }
            }, 1000);
        }

        // Complete Simple Tasks
        function completeTask(taskId, reward) {
            const taskElem = document.getElementById(taskId);
            if(taskElem) {
                // Open task URL
                window.open('https://youtube.com', '_blank');

                balance += reward;
                totalEarned += reward;
                tasksCompleted += 1;
                updateUI();

                // Disable completed task
                taskElem.classList.add('opacity-50', 'pointer-events-none');
                const btn = taskElem.querySelector('button');
                btn.innerText = "Completed";
                btn.classList.replace('bg-indigo-600', 'bg-slate-700');
            }
        }

        // Handle Withdrawal Request Form
        function handleWithdraw(event) {
            event.preventDefault();

            const method = document.querySelector('input[name="method"]:checked').value;
            const title = document.getElementById('accTitle').value.trim();
            const number = document.getElementById('accNumber').value.trim();
            const amount = parseInt(document.getElementById('withdrawAmount').value);
            const msgBox = document.getElementById('withdrawMsg');

            msgBox.classList.remove('hidden', 'bg-red-500/10', 'text-red-400', 'bg-emerald-500/10', 'text-emerald-400');

            if (amount < 500) {
                msgBox.classList.add('bg-red-500/10', 'text-red-400');
                msgBox.innerText = "Minimum withdrawal amount is Rs. 500";
                return;
            }

            if (amount > balance) {
                msgBox.classList.add('bg-red-500/10', 'text-red-400');
                msgBox.innerText = "Insufficient balance in your wallet!";
                return;
            }

            // Deduct balance
            balance -= amount;
            updateUI();

            // Clear inputs
            document.getElementById('withdrawForm').reset();

            // Show Success Notification
            msgBox.classList.add('bg-emerald-500/10', 'text-emerald-400');
            msgBox.innerText = `Withdrawal request of Rs. ${amount} submitted successfully via ${method}.`;

            // Append to history
            const historyList = document.getElementById('historyList');
            const emptyHist = document.getElementById('emptyHistory');
            if(emptyHist) emptyHist.remove();

            const historyItem = document.createElement('div');
            historyItem.className = "flex justify-between items-center p-2.5 rounded-lg bg-slate-900/60 border border-slate-800";
            historyItem.innerHTML = `
                <div>
                    <span class="font-bold text-white">${method}</span> (${number})
                    <p class="text-[10px] text-slate-500">${title}</p>
                </div>
                <div class="text-right">
                    <span class="font-bold text-amber-400">Rs. ${amount}</span>
                    <span class="block text-[10px] text-emerald-400">Pending</span>
                </div>
            `;
            historyList.prepend(historyItem);
        }
    </script>
